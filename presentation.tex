\begin{frame}{La pregunta}
  \begin{block}{No el tema, la pregunta}
    Cuando llega una IA capaz de hacer trabajo intelectual, ¿quiénes ganan y quiénes pierden según
    cuánto saben, y cómo cambia la respuesta si la IA puede trabajar sola o solo aconsejar?
  \end{block}
  \vfill
  \begin{itemize}
    \item La evidencia empírica se contradice: unos estudios encuentran que la IA ayuda más a los
      menos calificados, otros que beneficia a los más calificados.
    \item La apuesta del paper: ambas cosas pueden ser ciertas, y lo que decide es \emph{dentro de la
      organización} quién trabaja con quién.
  \end{itemize}
\end{frame}

\begin{frame}{El problema del agente}
  Conocimiento $z\in[0,1]$ con distribución $G$; dificultad $x\sim U[0,1]$, así que $z$ es la
  probabilidad de resolver un problema solo. Cada consulta cuesta $h$ del tiempo del solucionador:
  \[
    n(z)=\frac{1}{h(1-z)}\quad\text{trabajadores por solucionador.}
  \]
  Una firma de dos capas elige a su solucionador:
  \[
    \max_{s\in(z,1]}\ n(z)\,[\,s-w(z)\,]-w(s)
    \qquad\Longrightarrow\qquad
    \boxed{\,w'(m(z))=n(z)\,}
  \]
  \begin{block}{Léelo así}
    El salario marginal de un jefe \emph{es} el tamaño de su equipo: su conocimiento se aplica a
    $n(z)$ problemas a la vez. En una solución interior diferenciable, esta condición ayuda a explicar la convexidad salarial y la estratificación
    $\W\preceq\mathcal I\preceq\Sset$.
  \end{block}
\end{frame}

\begin{frame}{La IA, en tres supuestos}
  \begin{enumerate}
    \item \textbf{Escalabilidad}: el conocimiento $\zai$ se replica en todo el compute disponible.
    \item \textbf{Autonomía}: los agentes pueden ser co-trabajadores \emph{y} co-pilotos (en la
      Sección 6, solo co-pilotos).
    \item \textbf{Compute abundante} respecto del tiempo humano.
  \end{enumerate}
  \vfill
  El tercero fija el precio: algo de compute debe producir de forma independiente y esas firmas ganan
  cero, luego
  \[
    r^{*}=\zai \qquad\text{y}\qquad w^{*}(\zai)=\zai .
  \]
  \begin{block}{Primera consecuencia distributiva}
    El humano con conocimiento $\zai$ cae sobre la recta de 45° y \textbf{siempre pierde}: antes
    cobraba una renta, ahora compite con un sustituto perfecto de oferta elástica.
  \end{block}
\end{frame}

\begin{frame}{El resultado principal: condiciones y resultados}
  \small
  Supuestos del artículo: densidad $g$ continua y positiva, oportunidades de producción abundantes,
  competencia, $0<h<h_0$, $\zai\in[0,1)$ y cómputo abundante.
  \begin{block}{IA autónoma: Proposiciones 2 y 5}
    Equilibrio único y eficiente; $r^*=\zai$. Si $B$ y $T$ son los conjuntos de ganadores
    por debajo y por encima de $\zai$,
    \[
      B\ne\varnothing\iff\zai>\bar z_{AI},\quad
      \bar z_{AI}\in\operatorname{int}\W;
      \qquad T\ne\varnothing.
    \]
    Si la IA supervisa al tipo inferior, $w^*(0)=\zai(1-h)$; si el superior supervisa IA,
    $w^*(1)=1/h$. Son fórmulas de régimen.
  \end{block}
  \begin{block}{IA no autónoma: Proposición 6}
    Equilibrio único y eficiente, maximiza ingreso laboral, $r^\star=0$.
    Se usa si y solo si $\zai>w(0)$; si no, se preservan salarios y ocupaciones.
    El producto total es estrictamente menor que con IA autónoma.
  \end{block}
\end{frame}

\begin{frame}{Por qué: dos efectos que se pelean}
  \[
    w(z)=\underbrace{m(z)}_{\text{\emph{match}}}-\underbrace{\frac{w(m(z))}{n(z)}}_{\text{\emph{share}}}
  \]
  \begin{itemize}
    \item \textbf{Match, negativo abajo}: los de abajo quedan con jefes menos conocedores, porque
      entran solucionadores marginales nuevos.
    \item \textbf{Share, positivo}: el solucionador marginal pierde su renta y pasa a ganar lo que
      produciría solo, $C^{*}=\inf\Sset_p^{*}$; esa renta se reparte hacia abajo.
  \end{itemize}
  \vfill
  Arriba el share siempre gana, porque $h<h_0$ implica equipos grandes y el efecto se multiplica por
  $n(\cdot)$. Abajo depende de $\zai$, y de ahí sale el umbral.
\end{frame}

\begin{frame}{La trampa: capacidad y autonomía}
  \small
  \centering
  \begin{tabular}{@{}lcc@{}}
    \toprule
     & \textbf{Autónoma} & \textbf{No autónoma} \\ \midrule
    Renta del cómputo & $\zai$ & $0$ \\
    Salario inferior si usa IA & $\zai(1-h)$ & $\zai$ \\
    Ganadores estrictos abajo & $\zai>\bar z_{AI}$ & $\zai>w(0)$ \\
    Ganadores arriba vs.\ pre-IA & siempre & no necesariamente \\
    Producto total & mayor & menor \\ \bottomrule
  \end{tabular}
  \vfill
  \begin{alertblock}{No son dimensiones sustitutivas}
    Los umbrales son de capacidad. La autonomía cambia el conjunto de organizaciones
    factibles y el precio de la IA. El umbral autónomo no tiene la fórmula universal
    $w(0)/(1-h)$: depende del régimen en el que ocurra el cruce.
  \end{alertblock}
  \footnotesize Prop.\ 6: en un entorno de cero, $w^\star\ge\max\{w,w^*\}$ (estricto si se usa IA);
  cerca de uno, $w^\star\le w^*$ (estricto para $z\ne1$).
\end{frame}

\begin{frame}{Nuestra derivación: tres tipos}
  \small
  $0\le z_1<z_2<z_3<1$, masas $\mu_i>0$, $0<h<1$; $n_i=n(z_i)$,
  $n_a=n(a)$, cómputo $\mu$ y capacidad $a$.
  \[
    \begin{aligned}
      \text{(A0)}&\quad z_1\le a<z_3,\\
      \text{(A1)}&\quad \mu_1>n_1(\mu_2+\mu_3),\\
      \text{(A2)}&\quad \mu_2>n_2\mu_3,\\
      \text{(A3)}&\quad \mu>\mu_1/n_1+\mu_2/n_2+n_a(\mu_2+\mu_3).
    \end{aligned}
  \]
  \begin{itemize}
    \item Enumerar firmas puras: beneficio afín en las fracciones de tiempo.
    \item Salarios: máximos de ramas admisibles; no beneficio positivo, beneficio cero y vaciado.
    \item Umbral: $w_1^*>w_1\iff a>z_1/[1-h(1-z_1)]$.
    \item Con $a=z_2$ y ganancia estricta inferior, identidad
      $w_1^*-w_1=(w_2-w_2^*)/n_1$.
  \end{itemize}
  \footnotesize El discreto tiene independientes antes de IA; $z_1=0$ no recupera el continuo.
\end{frame}

\begin{frame}{Cálculos uniformes: el cruce depende del régimen}
  \small

  Continuidad salarial y matching implican
  \[
    z_2=\frac{a-ha^2/2}{1-h+ha},
    \qquad z_2-a=\frac{ha(1-3a/2)}{1-h+ha}.

\begin{frame}[fragile]{Lean: del enunciado matemático al teorema verificado}
  \textbf{1. La afirmación matemática.} Para $s<1$ y $x<y\le s$, el excedente del solucionador cae en el
  conocimiento del trabajador:
  \[
    n(y)\,(s-y)<n(x)\,(s-x),
    \qquad\text{porque}\qquad
    \frac{d}{dx}\frac{s-x}{1-x}=\frac{s-1}{(1-x)^{2}}<0 .
  \]
  \textbf{2. El enunciado en Lean y el paso central de la prueba.}
  \begin{semiverbatim}\footnotesize
theorem span_strict_anti \{x y s : \(\mathbb{R}\)\} (hx : x < y) (hy : y \(\le\) s) (hs : s < 1) :
    P.n y * (s - y) < P.n x * (s - x) := by
  have core : (s - y) * (P.h * (1 - x)) < (s - x) * (P.h * (1 - y)) := by
    nlinarith [mul_pos P.h_pos (mul_pos (sub_pos.mpr hx) (sub_pos.mpr hs))]
  \(\vdots\)
  exact lt_of_mul_lt_mul_right hmul (le_of_lt hprod)
  \end{semiverbatim}
\end{frame}

\begin{frame}{Lean: qué representa cada cosa y qué verifica}
  \begin{itemize}
    \item \texttt{P : Params} empaqueta $0<h<1$ y $0\le z_1<z_2<z_3<1$.
      El teorema usa además $x<y\le s<1$.
    \item \texttt{hs : s < 1} hace negativa la derivada. Para $s=1$ la
      desigualdad estricta falla.
    \item Multiplicamos por $h(1-x)h(1-y)>0$; usando
      $n(z)h(1-z)=1$, la diferencia algebraica es $h(y-x)(1-s)>0$.
  \end{itemize}
  \vfill
  \begin{block}{Evidencia de compilación}
    GitHub Actions \href{https://github.com/alejandroventuraventurameza-tech/ai-05-ide/actions/runs/45582527901}{45582527901}:
    doce teoremas discretos y once pruebas independientes del modelo continuo,
    con solo \texttt{propext}, \texttt{Classical.choice} y \texttt{Quot.sound}.
    Ninguna dependencia de \texttt{sorryAx}.
  \end{block}
\end{frame}

\begin{frame}{Dónde no le creí a la IA: dos errores discretos}
  \begin{block}{1. Una configuración de firma omitida}
    El salario del tipo 3 con IA no autónoma ignoraba que el solucionador podía
    contratar trabajadores del tipo 2. Beneficio cero en firmas activas no basta:
    hay que verificar que ninguna desviación es rentable.
  \end{block}
  \begin{block}{2. Una rama inadmisible}
    El test detectó muestras con $a<z_1$: el tipo 1 podía supervisar agentes de IA.
    Faltaba imponer (A0) o incluir una nueva configuración.
  \end{block}
  \vfill
  \textbf{Veredicto:} la derivación manuscrita y la enumeración de firmas
  corrigen los errores; las 20\,000 verificaciones no sustituyen la prueba.
\end{frame}
\begin{frame}{Dónde no le creí a la IA: dos errores formales}
  \small
  \begin{block}{Ingreso laboral: un objetivo sin presupuesto}
    $L(a)=\int w_a\,dG$. Cambiar solo $w_b(z)=L(a)+1$ conserva la factibilidad
    original y eleva $L(b)=L(a)+1$. Lean verifica
    \texttt{no\_labor\_income\_maximizer}.
  \end{block}
  \begin{block}{Unicidad: dominio del matching}
    En el modelo nadie con $z=1$ puede ser trabajador humano de un solucionador
    más conocedor. Alterar $m(1)$ conserva el equilibrio pero viola la igualdad
    de funciones en todo $[0,1]$.
    Lean verifica \texttt{no\_unique\_nonautonomous\_equilibrium}.
  \end{block}
  \textbf{Reparación propuesta:} comparar matching en su dominio activo y
  restringir $L(b)+r_b\mu\le Y(b)$. Con $r_a=0$ e identidad agregada,
  la eficiencia implica $L(b)\le Y(b)\le Y(a)=L(a)$.
  La identidad y la prueba completa permanecen pendientes.
\end{frame}

\begin{frame}{Cierre}
  \begin{itemize}
    \item La pregunta distributiva no tiene una sola palanca: hay \textbf{dos dimensiones}, capacidad y
      autonomía, y la lectura popular las colapsa en una.
    \item Restringir la autonomía favorece débilmente al extremo inferior frente a la autónoma y
      \emph{reduce estrictamente el producto}: esa es la tensión del artículo.
    \item Lo más frágil, a mi juicio: ``siempre hay ganadores arriba'' descansa en $h<h_0$. Con
      productores independientes el propio paper admite que puede fallar, y la versión discreta muestra
      por qué: los resultados dependen de los supuestos ocupacionales.
  \end{itemize}

\end{frame}
